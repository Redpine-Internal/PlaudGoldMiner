# Fase 2 — UI lê tabelas dos agentes n8n — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Repontar as rotas de leitura (lista+detalhe) das 3 telas da app das tabelas locais `app_*` (vazias) para as tabelas dos agentes n8n (`business_opportunities`/`social_posts`/`article_insights`), achatando o payload jsonb no shape que a UI já consome, sem mudar UI nem contrato de resposta.

**Architecture:** Cada rota vira um par _query crua (`pool.query`) + mapper puro_. Os mappers (`lib/n8n/mappers.ts`) achatam jsonb → cards e são testáveis sem banco. As rotas enriquecem `conversationTitle`/`conversationDate` num segundo SELECT em `conversations`. Read-only nesta fase: `PATCH` retorna 405; `analyze` intocado.

**Tech Stack:** Next.js App Router (route handlers), `pg` Pool via `@/lib/db` (`pool`), TypeScript, tsx para scripts de verificação (`node --env-file=.env --import tsx`), `node:assert/strict`.

## Contexto essencial (leia antes de começar)

**Fonte da verdade do banco** (introspecção cloud 2026-08-24, todas 0 linhas):
```
business_opportunities: id uuid, user_id text, meeting_ids text[], date_range_start date,
  date_range_end date, opportunities jsonb, raw_analysis text, created_at timestamptz
article_insights:       id uuid, user_id text, meeting_ids text[], title text, abstract_text text,
  key_findings jsonb, methodology_notes text, references_suggested jsonb, full_content text,
  focus_area text, created_at timestamptz
social_posts:           id uuid, user_id text, meeting_ids text[], platform text, content_type text,
  title text, body text, hashtags jsonb, image_prompt text, created_at timestamptz
```

**`pool` cru** é exportado de `@/lib/db` (`import { pool } from '@/lib/db'`). O driver `pg`
devolve `text[]` como array JS e `jsonb` como objeto/array JS já parseado — **não** re-parsear.

**Contrato de resposta atual** (o que a UI espera, NÃO mudar): `{ data: Card[], total: number }`.
As rotas de lista aceitam query params `limit` (default 50 em opps/contents, 20 em insights) e
filtros client-side (`status`, `type`/`platform`).

**Rodar um script tsx com env carregado:**
```bash

```
(`--env-file=.env` popula `process.env.MEETINGS_DATABASE_URL`, que `@/lib/db` lê.)

`tsconfig.json` já tem `"baseUrl": "."` + `"paths": { "@/*": ["./*"] }`, então `@/*` resolve sob tsx.

export function mapBusinessOpportunities(
  rows: BusinessOpportunityRow[]
): OpportunityCard[] {
  return rows.flatMap((row) => {
    const items = Array.isArray(row.opportunities)
      ? (row.opportunities as Record<string, unknown>[])
      : [];
    const conversationId = row.meeting_ids?.[0] ?? null;
    return items.map((it, i) => ({
      id: (it.id as string) ?? `${row.id}:${i}`,
      title: (it.title as string) ?? 'Sem título',
      pain: (it.pain as string) ?? '',
      context: (it.context as string) ?? null,
      score: Number(it.score ?? 0),
      type: (it.type as string) ?? 'produto',
      status: (it.status as string) ?? 'nova',
      notes: null,
      tags: null,
      conversationId,
      createdAt: row.created_at,

  });
}

import { pool } from '@/lib/db';

// Preenche conversationTitle/conversationDate nos cards a partir da view
// `conversations` (que enxerga `meetings`). Faz UM SELECT com id = ANY($1).
// Cards com conversationId null ficam sem título (undefined). Mutação in-place
// evitada: retorna cópia enriquecida.
export async function enrichWithConversation<
  T extends { conversationId: string | null }
>(cards: T[]): Promise<(T & { conversationTitle?: string | null; conversationDate?: string | null })[]> {
  const ids = [...new Set(cards.map((c) => c.conversationId).filter((x): x is string => !!x))];
  if (ids.length === 0) return cards.map((c) => ({ ...c }));

  const res = await pool.query<{ id: string; title: string; date: string }>(
    `SELECT id, title, date::text AS date FROM conversations WHERE id = ANY($1)`,
    [ids]
  );
  const byId = new Map(res.rows.map((r) => [r.id, r]));
  return cards.map((c) => {
    const conv = c.conversationId ? byId.get(c.conversationId) : undefined;
    return { ...c, conversationTitle: conv?.title ?? null, conversationDate: conv?.date ?? null };
  });
}
```

import { NextRequest, NextResponse } from 'next/server';
import { pool } from '@/lib/db';
import { mapBusinessOpportunities, type BusinessOpportunityRow } from '@/lib/n8n/mappers';
import { enrichWithConversation } from '@/lib/n8n/enrich';

export async function GET(request: NextRequest) {
  try {
    const searchParams = request.nextUrl.searchParams;
    const limit = parseInt(searchParams.get('limit') || '50', 10);
    const status = searchParams.get('status');
    const type = searchParams.get('type');

    const res = await pool.query<BusinessOpportunityRow>(
      `SELECT id, meeting_ids, opportunities, created_at
         FROM business_opportunities
        ORDER BY created_at DESC
        LIMIT $1`,
      [limit]
    );

    const cards = await enrichWithConversation(mapBusinessOpportunities(res.rows));

    let filtered = cards;
    if (status) filtered = filtered.filter((o) => o.status === status);
    if (type) filtered = filtered.filter((o) => o.type === type);

    return NextResponse.json({ data: filtered, total: filtered.length });
  } catch (error) {
    console.error('Error fetching opportunities:', error);
    return NextResponse.json({ error: 'Failed to fetch opportunities' }, { status: 500 });
  }
}

import { NextRequest, NextResponse } from 'next/server';
import { pool } from '@/lib/db';
import { mapBusinessOpportunities, type BusinessOpportunityRow } from '@/lib/n8n/mappers';
import { enrichWithConversation } from '@/lib/n8n/enrich';

// id sintético = `${rowId}:${index}`; id próprio do item também é aceito (busca por índice
// falha → varre a linha inteira). Estratégia: separar rowId antes do ':' e buscar essa linha.
export async function GET(
  request: NextRequest,
  { params }: { params: Promise<{ id: string }> }
) {
  try {
    const { id } = await params;
    const rowId = id.includes(':') ? id.slice(0, id.lastIndexOf(':')) : id;

    const res = await pool.query<BusinessOpportunityRow>(
      `SELECT id, meeting_ids, opportunities, created_at
         FROM business_opportunities WHERE id = $1 LIMIT 1`,
      [rowId]
    );
    if (res.rowCount === 0) {
      return NextResponse.json({ error: 'Opportunity not found' }, { status: 404 });
    }

    const cards = await enrichWithConversation(mapBusinessOpportunities(res.rows));
    const card = cards.find((c) => c.id === id);
    if (!card) {
      return NextResponse.json({ error: 'Opportunity not found' }, { status: 404 });
    }
    return NextResponse.json({ data: card });
  } catch (error) {
    console.error('Error fetching opportunity:', error);
    return NextResponse.json({ error: 'Failed to fetch opportunity' }, { status: 500 });
  }
}

// Read-only nesta fase: os dados são gerados pelo agente n8n (sem estado editável).

}

    return NextResponse.json({ error: 'Failed to fetch insight' }, { status: 500 });
  }
}

export async function PATCH() {
  return NextResponse.json(
    { error: 'Edição desabilitada nesta fase (dados read-only do n8n)' },
    { status: 405 }
  );
}
```

import { pool } from '@/lib/db';

function req(url: string) {
  // NextRequest aceita um Request padrão; searchParams via URL.
  return new Request(url) as unknown as import('next/server').NextRequest;
}

async function main() {
  // ---- seed ----
  await pool.query(
    `INSERT INTO business_opportunities (id, meeting_ids, opportunities, created_at)
     VALUES ($1, ARRAY[]::text[], $2::jsonb, now())`,
    [BO_ID, JSON.stringify([
      { title: 'OpX', pain: 'dorX', score: 90, type: 'produto' },
      { title: 'OpY', pain: 'dorY', score: 10, type: 'servico' },
    ])]
  );
  await pool.query(
    `INSERT INTO social_posts (id, meeting_ids, platform, content_type, title, body, hashtags, created_at)
     VALUES ($1, ARRAY[]::text[], 'linkedin', 'post', 'T', 'B', $2::jsonb, now())`,
    [SP_ID, JSON.stringify(['#x'])]
  );
  await pool.query(
    `INSERT INTO article_insights (id, meeting_ids, title, abstract_text, focus_area, created_at)
     VALUES ($1, ARRAY[]::text[], 'Art', 'resumo', 'seguranca', now())`,
    [AI_ID]
  );

  try {
    // ---- lista de oportunidades: achatamento 1→2 ----
    const oppRes = await oppList(req('http://x/api/opportunities?limit=50'));
    const oppBody = await oppRes.json();
    const mine = oppBody.data.filter((o: any) => o.id.startsWith(BO_ID));
    assert.equal(mine.length, 2, '1 linha com 2 itens → 2 cards');
    assert.equal(mine[0].id, `${BO_ID}:0`);
    assert.equal(mine[0].title, 'OpX');

    // ---- filtro por type ----

    const fb = await oppFiltered.json();

    assert.equal(dBody.data.title, 'OpY');

      params: Promise.resolve({ id: `${BO_ID}:9` }),
    });
    assert.equal(missRes.status, 404, 'índice fora do array → 404');

    const patchRes = await oppPatch();

    assert.equal(ins.pattern, 'seguranca');

    // ---- cleanup (sempre) ----
    await pool.query('DELETE FROM business_opportunities WHERE id = $1', [BO_ID]);
    await pool.query('DELETE FROM social_posts WHERE id = $1', [SP_ID]);
    await pool.query('DELETE FROM article_insights WHERE id = $1', [AI_ID]);
    await pool.end();
  }
}

main().catch(async (e) => {
  console.error(e);
  process.exit(1);
});
